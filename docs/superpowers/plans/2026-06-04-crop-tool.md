# Crop Tool Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a `/crop` page that lets users upload one or more images, optionally composite a Steam profile background, then slice and save the results to a local folder via the File System Access API.

**Architecture:** A new `CropView.vue` handles the UI (type selector, background URL input, file picker, output folder picker, crop button). All canvas-based image processing lives in `src/utils/crop.ts` so it can be reasoned about and tested independently of Vue. The router gets a new lazy-loaded `/crop` route.

**Tech Stack:** Vue 3 + TypeScript, Canvas API, File System Access API, Vite, vue-router 5

```ts
export interface CropRegion {
  /** X offset into the 1920px background image */
  bgX: number
  /** Y offset into the 1920px background image */
  bgY: number
  /** Width to crop from background */
  bgW: number

}

export interface SliceOutput {
  /** Sub-path within the output directory, e.g. "left/foo.png" */
  subPath: string

}

  1: { bgX: 494, bgY: 256, bgW: 630, finalW: 630 },
```

### Task 3: Implement `buildComposite` — scale + composite onto background

**Files:**
- Modify: `src/utils/crop.ts`

- [ ] **Step 1: Add `loadImage` helper and `buildComposite` function**

Append to `src/utils/crop.ts`:

```ts
function loadImage(src: string): Promise<HTMLImageElement> {
  return new Promise((resolve, reject) => {
    const img = new Image()
    img.crossOrigin = 'anonymous'
    img.onload = () => resolve(img)
    img.onerror = reject
    img.src = src
  })
}

/**
 * Builds a composited canvas for one user image:
 * 1. Scale user image to targetW (maintaining aspect ratio)
 * 2. If bgUrl provided: crop bg region, draw on black canvas, overlay user image
 * 3. If no bgUrl: draw user image on black canvas
 */
export async function buildComposite(
  userFile: File,
  type: ShowcaseType,
  bgUrl: string | null,
): Promise<HTMLCanvasElement> {
  const region = BG_REGIONS[type]
  const userImg = await loadImage(URL.createObjectURL(userFile))

  // Scale user image to targetW preserving aspect ratio
  const scale = region.finalW / userImg.naturalWidth
  const scaledW = region.finalW
  const scaledH = Math.round(userImg.naturalHeight * scale)

  const canvas = document.createElement('canvas')
  canvas.width = scaledW
  canvas.height = scaledH
  const ctx = canvas.getContext('2d')!

  // Black base
  ctx.fillStyle = '#000000'
  ctx.fillRect(0, 0, scaledW, scaledH)

  // Background layer
  if (bgUrl) {
    const bgImg = await loadImage(bgUrl)
    // bgImg is the full 1920px Steam background; crop the region for this type
    const bgCropH = bgImg.naturalHeight - region.bgY
    // Draw cropped bg scaled to fit our canvas width; height proportional
    const bgDestH = Math.round((bgCropH / region.bgW) * scaledW)
    ctx.drawImage(
      bgImg,
      region.bgX, region.bgY, region.bgW, bgCropH,
      0, 0, scaledW, Math.min(bgDestH, scaledH),
    )
  }

  // User image on top
  ctx.drawImage(userImg, 0, 0, scaledW, scaledH)

  URL.revokeObjectURL(userImg.src)
  return canvas
}
```

### Task 4: Implement `sliceCanvas` — split composited canvas per showcase type

**Files:**
- Modify: `src/utils/crop.ts`

- [ ] **Step 1: Add `sliceCanvas` function**

Append to `src/utils/crop.ts`:

```ts
/**
 * Slices a composited canvas according to showcase type rules.
 * Returns one or more {subPath, canvas} pairs ready for saving.
 *
 * @param composite  Output of buildComposite
 * @param type       Showcase type
 * @param filename   Original file name (used as the leaf filename in subPath)
 * @param trimRight  Type 0 only: trim 70px from bottom of right slice
 */
export function sliceCanvas(
  composite: HTMLCanvasElement,
  type: ShowcaseType,
  filename: string,
  trimRight = false,
): SliceOutput[] {
  function extractRegion(
    src: HTMLCanvasElement,
    sx: number, sy: number, sw: number, sh: number,
  ): HTMLCanvasElement {
    const c = document.createElement('canvas')
    c.width = sw
    c.height = sh
    c.getContext('2d')!.drawImage(src, sx, sy, sw, sh, 0, 0, sw, sh)
    return c
  }

  const h = composite.height

  if (type === 0) {
    // Left: x=0, w=506, full height
    const left = extractRegion(composite, 0, 0, 506, h)
    // Right: x=515, w=100, height optionally trimmed by 70px
    const rightH = trimRight ? Math.max(0, h - 70) : h
    const right = extractRegion(composite, 515, 0, 100, rightH)
    return [
      { subPath: `left/${filename}`, canvas: left },
      { subPath: `right/${filename}`, canvas: right },
    ]
  }

  if (type === 1) {
    // No slicing — output the full composite
    return [{ subPath: filename, canvas: composite }]
  }

  // type === 2: 5 equal slices, w=122, step=126
  return Array.from({ length: 5 }, (_, i) => ({
    subPath: `${i + 1}/${filename}`,
    canvas: extractRegion(composite, i * 126, 0, 122, h),
  }))
}
```

```ts
function canvasToBlob(canvas: HTMLCanvasElement, type = 'image/png'): Promise<Blob> {
  return new Promise((resolve, reject) => {
    canvas.toBlob((b) => b ? resolve(b) : reject(new Error('toBlob failed')), type)
  })
}

/**
 * Writes all slices to the given FileSystemDirectoryHandle.
 * Sub-directories (e.g. "left/", "1/") are created automatically.
 */
export async function saveSlices(
  slices: SliceOutput[],
  dir: FileSystemDirectoryHandle,
): Promise<void> {
  for (const { subPath, canvas } of slices) {
    const parts = subPath.split('/')
    let currentDir = dir
    // Create intermediate directories
    for (const part of parts.slice(0, -1)) {
      currentDir = await currentDir.getDirectoryHandle(part, { create: true })
    }
    const leafName = parts[parts.length - 1]
    const fileHandle = await currentDir.getFileHandle(leafName, { create: true })
    const writable = await fileHandle.createWritable()
    const blob = await canvasToBlob(canvas)
    await writable.write(blob)
    await writable.close()
  }
}
```

```vue
<script setup lang="ts">
import { ref } from 'vue'
import { buildComposite, sliceCanvas, saveSlices } from '@/utils/crop'
import type { ShowcaseType } from '@/utils/crop'

const type = ref<ShowcaseType>(0)
const bgUrl = ref('')
const files = ref<File[]>([])
const trimRight = ref(false)
const dirHandle = ref<FileSystemDirectoryHandle | null>(null)
const status = ref('')
const running = ref(false)

function onFileChange(e: Event) {
  const input = e.target as HTMLInputElement
  files.value = input.files ? Array.from(input.files) : []
}

async function pickFolder() {
  try {
    dirHandle.value = await window.showDirectoryPicker({ mode: 'readwrite' })
    status.value = `已選擇資料夾：${dirHandle.value.name}`
  } catch {
    // user cancelled
  }
}

async function run() {
  if (!files.value.length) { status.value = '請先選擇圖片'; return }
  if (!dirHandle.value) { status.value = '請先選擇輸出資料夾'; return }
  running.value = true
  status.value = ''
  try {
    for (let i = 0; i < files.value.length; i++) {
      const file = files.value[i]
      status.value = `處理中 ${i + 1} / ${files.value.length}：${file.name}`
      const composite = await buildComposite(file, type.value, bgUrl.value.trim() || null)
      const slices = sliceCanvas(composite, type.value, file.name, trimRight.value)
      await saveSlices(slices, dirHandle.value!)
    }
    status.value = `完成！共處理 ${files.value.length} 張圖片`
  } catch (err) {
    status.value = `錯誤：${(err as Error).message}`
  } finally {
    running.value = false
  }
}
</script>
```
